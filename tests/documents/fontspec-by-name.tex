% engines: xelatex lualatex
% prefetch: lualatex
% Fonts selected by family name, not installed beforehand: XeTeX asks
% kpathsea (MennoTeX XeTeX patch), luaotfload probes the name as a TFM;
% both reach mtx's font-name index, which installs the package. fontspec
% sends LuaTeX `name:` requests, which luaotfload resolves (and gives up
% on) before any kpathsea lookup, so LuaLaTeX needs `mtx prefetch` to
% succeed on the first run; the failed run installs the fonts for the next.
\documentclass{article}
\usepackage{fontspec}
\setmainfont{TeX Gyre Pagella}
\setsansfont{TeX Gyre Heros}
\setmonofont{TeX Gyre Cursor}
\begin{document}
Pagella with \textbf{bold} and \textit{italic}, \textsf{Heros sans}, \texttt{Cursor mono}.
\end{document}
