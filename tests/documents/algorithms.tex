\caption{Euclid's algorithm}\label{alg:euclid}
\begin{algorithmic}[1]
\Procedure{Euclid}{$a,b$}
  \While{$b \neq 0$}
    \State $(a, b) \gets (b, a \bmod b)$
  \EndWhile
  \State \Return $a$
\EndProcedure
\end{algorithmic}
\end{algorithm}
